data class Appointment(
    val id: Long = System.currentTimeMillis(),
    val title: String,
    val category: String,
    val duration: String,
    val location: String,
    val timeInfo: String,
    val invitedFriends: List<String>
)

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            PotatoMeetTheme {
                val currentScreen = remember { mutableStateOf("login") }
                val userNickname = remember { mutableStateOf("") }

                val appointmentName = remember { mutableStateOf("") }
                val category = remember { mutableStateOf("") }
                val duration = remember { mutableStateOf("") }
                val location = remember { mutableStateOf("") }
                val availableTimeInput = remember { mutableStateOf("") }
                val selectedFriendsForNew = remember { mutableStateMapOf<String, Boolean>() }
                
                val coordinatingAppointments = remember { mutableStateListOf<Appointment>() }

                BackHandler(enabled = currentScreen.value != "login") {
                    when (currentScreen.value) {
                        "signup" -> currentScreen.value = "login"
                        "home" -> finish()
                        "create_step1" -> currentScreen.value = "home"
                        "create_step2" -> currentScreen.value = "create_step1"
                        "create_step3" -> currentScreen.value = "create_step2"
                    }
                }

                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = Color(0xFFFFFCF3)
                ) {
                    when (currentScreen.value) {
                        "login" -> {
                            LoginScreen(
                                onLoginClick = { currentScreen.value = "home" },
                                onSignUpClick = { currentScreen.value = "signup" }
                            )
                        }
                        "signup" -> {
                            SignUpScreen(
                                onCompleteClick = { enteredName ->
                                    if (enteredName.isNotBlank()) {
                                        userNickname.value = enteredName.trim()
                                    }
                                    currentScreen.value = "home"
                                }
                            )
                        }
                        "home" -> {
                            HomeScreen(
                                nickname = userNickname.value,
                                coordinatingList = coordinatingAppointments,
                                onLogoutClick = {
                                    userNickname.value = ""
                                    currentScreen.value = "login"
                                },
                                onCreateAppointmentClick = {
                                    currentScreen.value = "create_step1"
                                },
                                onDeleteAppointment = { appt ->
                                    coordinatingAppointments.remove(appt)
                                },
                                onUpdateAppointment = { updatedAppt ->
                                    val index = coordinatingAppointments.indexOfFirst { it.id == updatedAppt.id }
                                    if (index != -1) {
                                        coordinatingAppointments[index] = updatedAppt
                                    }
                                }
                            )
                        }
                        "create_step1" -> {
                            CreateAppointmentStep1Screen(
                                appointmentName = appointmentName.value,
                                onAppointmentNameChange = { appointmentName.value = it },
                                category = category.value,
                                onCategoryChange = { category.value = it },
                                duration = duration.value,
                                onDurationChange = { duration.value = it },
                                location = location.value,
                                onLocationChange = { location.value = it },
                                onBackClick = { currentScreen.value = "home" },
                                onNextClick = { currentScreen.value = "create_step2" },
                                onStepClick = { step -> currentScreen.value = step }
                            )
                        }
                        "create_step2" -> {
                            CreateAppointmentStep2Screen(
                                availableTimeInput = availableTimeInput.value,
                                onAvailableTimeInputChange = { availableTimeInput.value = it },
                                onBackClick = { currentScreen.value = "create_step1" },
                                onNextClick = { currentScreen.value = "create_step3" },
                                onStepClick = { step -> currentScreen.value = step }
                            )
                        }
                        "create_step3" -> {
                            CreateAppointmentStep3Screen(
                                selectedFriendsMap = selectedFriendsForNew,
                                onBackClick = { currentScreen.value = "create_step2" },
                                onStepClick = { step -> currentScreen.value = step },
                                onComplete = { friends ->
                                    val newAppt = Appointment(
                                        title = appointmentName.value.ifBlank { "제목 없는 약속" },
                                        category = category.value.ifBlank { "기타" },
                                        duration = duration.value.ifBlank { "시간 미정" },
                                        location = location.value.ifBlank { "장소 미정" },
                                        timeInfo = availableTimeInput.value.ifBlank { "일정 조율 중" },
                                        invitedFriends = friends
                                    )
                                    coordinatingAppointments.add(newAppt)

                                    appointmentName.value = ""
                                    category.value = ""
                                    duration.value = ""
                                    location.value = ""
                                    availableTimeInput.value = ""
                                    selectedFriendsForNew.clear()
                                    currentScreen.value = "home"
                                }
                            )
                        }
                    }
                }
            }
        }
    }
}

// 2. 회원가입 화면
@Composable
fun SignUpScreen(onCompleteClick: (String) -> Unit) {
    var nickname by remember { mutableStateOf("") }
    val keyboardController = LocalSoftwareKeyboardController.current

    Column(
        modifier = Modifier.fillMaxSize().padding(horizontal = 30.dp).verticalScroll(rememberScrollState()),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Image(painter = painterResource(id = R.drawable.potato_logo), contentDescription = "Logo", modifier = Modifier.size(180.dp))
        Spacer(modifier = Modifier.height(12.dp))
        Text(text = "함께하는 일정, 더 쉬운 약속", fontSize = 15.sp, color = Color(0xFF666666))
        Spacer(modifier = Modifier.height(35.dp))
        OutlinedTextField(
            value = nickname,
            onValueChange = { nickname = it },
            placeholder = { Text("사용할 닉네임을 입력하세요", color = Color(0xFFAAAAAA)) },
            singleLine = true,
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Text, imeAction = ImeAction.Done),
            keyboardActions = KeyboardActions(onDone = { keyboardController?.hide(); onCompleteClick(nickname) }),
            modifier = Modifier.fillMaxWidth().height(56.dp),
            shape = RoundedCornerShape(28.dp),
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = Color(0xFFFFBA00),
                unfocusedBorderColor = Color(0xFFE0E0E0),
                focusedContainerColor = Color.White,
                unfocusedContainerColor = Color.White
            )
        )
        Spacer(modifier = Modifier.height(20.dp))
        Button(
            onClick = { onCompleteClick(nickname) },
            modifier = Modifier.fillMaxWidth().height(54.dp),
            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFFBA00)),
            shape = RoundedCornerShape(27.dp)
        ) {
            Text(text = "완료", fontSize = 17.sp, fontWeight = FontWeight.Bold, color = Color(0xFF3C1E1E))
        }
    }
}

// 3. 홈 화면 및 상세 다이얼로그
@Composable
fun HomeScreen(
    nickname: String,
    coordinatingList: List<Appointment>,
    onLogoutClick: () -> Unit,
    onCreateAppointmentClick: () -> Unit,
    onDeleteAppointment: (Appointment) -> Unit,
    onUpdateAppointment: (Appointment) -> Unit
) {
    var selectedAppointment by remember { mutableStateOf<Appointment?>(null) }
    var isEditing by remember { mutableStateOf(false) }

    Column(
        modifier = Modifier.fillMaxSize().padding(24.dp)
    ) {
        Spacer(modifier = Modifier.height(20.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Image(painter = painterResource(id = R.drawable.potato_logo), contentDescription = "Logo", modifier = Modifier.size(54.dp))
                Spacer(modifier = Modifier.width(12.dp))
                Text(
                    text = buildAnnotatedString {
                        withStyle(SpanStyle(color = Color(0xFF4A3B32), fontWeight = FontWeight.ExtraBold)) { append("potato ") }
                        withStyle(SpanStyle(color = Color(0xFFFF9800), fontWeight = FontWeight.ExtraBold)) { append("meet") }
                    },
                    fontSize = 30.sp
                )
            }
            TextButton(onClick = onLogoutClick) {
                Text(text = "로그아웃", fontSize = 12.sp, color = Color(0xFF888888))
            }
        }

        Spacer(modifier = Modifier.height(24.dp))

        val greetingText = if (nickname.isNotBlank()) "안녕하세요, ${nickname}님! 👋" else "안녕하세요! 👋"
        Text(text = greetingText, fontSize = 22.sp, fontWeight = FontWeight.Bold, color = Color(0xFF333333))
        Text(text = "오늘도 좋은 하루 보내세요!", fontSize = 14.sp, color = Color(0xFF777777))

        Spacer(modifier = Modifier.height(20.dp))

        Button(
            onClick = onCreateAppointmentClick,
            modifier = Modifier.fillMaxWidth().height(52.dp),
            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFFBA00)),
            shape = RoundedCornerShape(26.dp)
        ) {
            Text(text = "+ 약속 만들기", fontSize = 16.sp, fontWeight = FontWeight.Bold, color = Color(0xFF3C1E1E))
        }

        Spacer(modifier = Modifier.height(24.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(text = "다가오는 약속", fontSize = 16.sp, fontWeight = FontWeight.Bold, color = Color(0xFF333333))
            Text(text = "전체보기 >", fontSize = 13.sp, color = Color(0xFF888888))
        }
        Spacer(modifier = Modifier.height(10.dp))
        Box(
            modifier = Modifier.fillMaxWidth().height(90.dp).clip(RoundedCornerShape(16.dp)).background(Color.White),
            contentAlignment = Alignment.Center
        ) {
            Text(text = "약속이 없습니다.", fontSize = 14.sp, color = Color(0xFFAAAAAA))
        }

        Spacer(modifier = Modifier.height(24.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(text = "조율 중인 약속", fontSize = 16.sp, fontWeight = FontWeight.Bold, color = Color(0xFF333333))
            Text(text = "전체보기 >", fontSize = 13.sp, color = Color(0xFF888888))
        }
        Spacer(modifier = Modifier.height(10.dp))

        if (coordinatingList.isEmpty()) {
            Box(
                modifier = Modifier.fillMaxWidth().height(90.dp).clip(RoundedCornerShape(16.dp)).background(Color.White),
                contentAlignment = Alignment.Center
            ) {
                Text(text = "약속이 없습니다.", fontSize = 14.sp, color = Color(0xFFAAAAAA))
            }
        } else {
            LazyColumn(
                verticalArrangement = Arrangement.spacedBy(10.dp),
                modifier = Modifier.fillMaxWidth().height(180.dp)
            ) {
                items(coordinatingList) { appt ->
                    Card(
                        modifier = Modifier
                            .fillMaxWidth()
                            .clickable {
                                selectedAppointment = appt
                                isEditing = false
                            },
                        colors = CardDefaults.cardColors(containerColor = Color.White),
                        shape = RoundedCornerShape(16.dp),
                        elevation = CardDefaults.cardElevation(defaultElevation = 1.dp)
                    ) {
                        Column(modifier = Modifier.padding(16.dp)) {
                            Row(
                                modifier = Modifier.fillMaxWidth(),
                                horizontalArrangement = Arrangement.SpaceBetween,
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                Text(text = appt.title, fontSize = 15.sp, fontWeight = FontWeight.Bold, color = Color(0xFF333333))
                                Surface(
                                    color = Color(0xFFFFECB3),
                                    shape = RoundedCornerShape(12.dp)
                                ) {
                                    Text(
                                        text = "조율 중",
                                        fontSize = 11.sp,
                                        fontWeight = FontWeight.Bold,
                                        color = Color(0xFFE65100),
                                        modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp)
                                    )
                                }
                            }
                            Spacer(modifier = Modifier.height(6.dp))
                            Text(text = "📅 ${appt.timeInfo}", fontSize = 13.sp, color = Color(0xFF666666))
                            Text(text = "📍 ${appt.location} (${appt.category})", fontSize = 13.sp, color = Color(0xFF666666))
                        }
                    }
                }
            }
        }
    }

    // 💡 상세 / 수정 다이얼로그 (닫기 버튼 오른쪽 고정 및 휴지통 옆 '삭제' 빨간색 표시)
    selectedAppointment?.let { appt ->
        var editTitle by remember { mutableStateOf(appt.title) }
        var editLocation by remember { mutableStateOf(appt.location) }
        var editTime by remember { mutableStateOf(appt.timeInfo) }

        val allFriends = listOf("김민수", "이지훈", "박영희", "이철수")
        val editSelectedFriends = remember {
            mutableStateMapOf<String, Boolean>().apply {
                allFriends.forEach { friend ->
                    put(friend, appt.invitedFriends.contains(friend))
                }
            }
        }

        var showDeleteConfirm by remember { mutableStateOf(false) }
        var showCloseConfirm by remember { mutableStateOf(false) }

        AlertDialog(
            onDismissRequest = {
                if (isEditing) {
                    showCloseConfirm = true
                } else {
                    selectedAppointment = null
                }
            },
            containerColor = Color.White,
            title = {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = if (isEditing) "약속 수정하기" else "약속 상세 정보",
                        fontWeight = FontWeight.Bold,
                        color = Color(0xFF333333)
                    )
                    // 🗑️ 휴지통 아이콘 옆에 빨간색 '삭제' 문구 배치
                    TextButton(onClick = { showDeleteConfirm = true }) {
                        Text(text = "🗑️ 삭제", fontSize = 14.sp, fontWeight = FontWeight.Bold, color = Color.Red)
                    }
                }
            },
            text = {
                Column(
                    modifier = Modifier.verticalScroll(rememberScrollState()),
                    verticalArrangement = Arrangement.spacedBy(10.dp)
                ) {
                    if (!isEditing) {
                        Text(text = "📌 약속 이름: ${appt.title}", fontSize = 15.sp, fontWeight = FontWeight.Bold, color = Color(0xFF333333))
                        Text(text = "🏷️ 카테고리: ${appt.category}", fontSize = 14.sp, color = Color(0xFF666666))
                        Text(text = "⏱️ 소요 시간: ${appt.duration}", fontSize = 14.sp, color = Color(0xFF666666))
                        Text(text = "📍 장소: ${appt.location}", fontSize = 14.sp, color = Color(0xFF666666))
                        Text(text = "📅 시간: ${appt.timeInfo}", fontSize = 14.sp, color = Color(0xFF666666))
                        if (appt.invitedFriends.isNotEmpty()) {
                            Text(text = "👥 초대된 친구: ${appt.invitedFriends.joinToString(", ")}", fontSize = 14.sp, color = Color(0xFF666666))
                        }
                    } else {
                        OutlinedTextField(
                            value = editTitle,
                            onValueChange = { editTitle = it },
                            label = { Text("약속 이름") },
                            singleLine = true,
                            modifier = Modifier.fillMaxWidth()
                        )
                        OutlinedTextField(
                            value = editLocation,
                            onValueChange = { editLocation = it },
                            label = { Text("장소") },
                            singleLine = true,
                            modifier = Modifier.fillMaxWidth()
                        )
                        OutlinedTextField(
                            value = editTime,
                            onValueChange = { editTime = it },
                            label = { Text("조율 시간") },
                            singleLine = true,
                            modifier = Modifier.fillMaxWidth()
                        )

                        Spacer(modifier = Modifier.height(4.dp))
                        Text(text = "👥 친구 추가 및 관리", fontSize = 13.sp, fontWeight = FontWeight.Bold, color = Color(0xFF555555))
                        
                        Column(modifier = Modifier.height(130.dp).verticalScroll(rememberScrollState())) {
                            allFriends.forEach { friend ->
                                val isChecked = editSelectedFriends[friend] ?: false
                                Row(
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .clickable { editSelectedFriends[friend] = !isChecked }
                                        .padding(vertical = 4.dp),
                                    horizontalArrangement = Arrangement.SpaceBetween,
                                    verticalAlignment = Alignment.CenterVertically
                                ) {
                                    Text(text = friend, fontSize = 14.sp, color = Color(0xFF333333))
                                    Checkbox(
                                        checked = isChecked,
                                        onCheckedChange = { editSelectedFriends[friend] = it },
                                        colors = CheckboxDefaults.colors(checkedColor = Color(0xFFFFBA00))
                                    )
                                }
                            }
                        }
                    }
                }
            },
            // 왼쪽 액션 버튼 (기본: 수정, 수정모드: 저장)
            confirmButton = {
                if (!isEditing) {
                    Button(
                        onClick = { isEditing = true },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFFBA00))
                    ) {
                        Text(text = "수정", color = Color(0xFF3C1E1E), fontWeight = FontWeight.Bold)
                    }
                } else {
                    Button(
                        onClick = {
                            val newFriendsList = allFriends.filter { editSelectedFriends[it] == true }
                            val updated = appt.copy(
                                title = editTitle,
                                location = editLocation,
                                timeInfo = editTime,
                                invitedFriends = newFriendsList
                            )
                            onUpdateAppointment(updated)
                            selectedAppointment = null
                            isEditing = false
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFFBA00))
                    ) {
                        Text(text = "저장", color = Color(0xFF3C1E1E), fontWeight = FontWeight.Bold)
                    }
                }
            },
            // 오른쪽 고정 닫기 버튼
            dismissButton = {
                TextButton(
                    onClick = {
                        if (isEditing) {
                            showCloseConfirm = true
                        } else {
                            selectedAppointment = null
                        }
                    }
                ) {
                    Text(text = "닫기", color = Color(0xFF888888), fontWeight = FontWeight.Bold)
                }
            }
        )

        if (showDeleteConfirm) {
            AlertDialog(
                onDismissRequest = { showDeleteConfirm = false },
                title = { Text(text = "약속 삭제", fontWeight = FontWeight.Bold) },
                text = { Text(text = "정말 약속을 삭제하시겠습니까?") },
                confirmButton = {
                    Button(
                        onClick = {
                            onDeleteAppointment(appt)
                            showDeleteConfirm = false
                            selectedAppointment = null
                            isEditing = false
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = Color.Red)
                    ) {
                        Text(text = "삭제", color = Color.White)
                    }
                },
                dismissButton = {
                    TextButton(onClick = { showDeleteConfirm = false }) {
                        Text(text = "취소")
                    }
                }
            )
        }

        if (showCloseConfirm) {
            AlertDialog(
                onDismissRequest = { showCloseConfirm = false },
                title = { Text(text = "변경사항 확인", fontWeight = FontWeight.Bold) },
                text = { Text(text = "저장하시겠습니까?") },
                confirmButton = {
                    Button(
                        onClick = {
                            val newFriendsList = allFriends.filter { editSelectedFriends[it] == true }
                            val updated = appt.copy(
                                title = editTitle,
                                location = editLocation,
                                timeInfo = editTime,
                                invitedFriends = newFriendsList
                            )
                            onUpdateAppointment(updated)
                            showCloseConfirm = false
                            selectedAppointment = null
                            isEditing = false
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFFBA00))
                    ) {
                        Text(text = "저장", color = Color(0xFF3C1E1E), fontWeight = FontWeight.Bold)
                    }
                },
                dismissButton = {
                    TextButton(
                        onClick = {
                            showCloseConfirm = false
                            selectedAppointment = null
                            isEditing = false
                        }
                    ) {
                        Text(text = "아니오 (그냥 닫기)", color = Color.Red)
                    }
                }
            )
        }
    }
}

// 4. 약속 만들기 - Step 1: 약속 정보 입력 (스텝 클릭 네비게이션 연동)
@Composable
fun CreateAppointmentStep1Screen(
    appointmentName: String,
